cmake_minimum_required(VERSION 3.22)
project(SpeedBreakerAndroid VERSION 0.1.38 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SB_ANDROID_BUILD_JOBS "0" CACHE STRING "Optional cap on native compiler jobs (0 keeps Ninja default)")
if(SB_ANDROID_BUILD_JOBS MATCHES "^[1-9][0-9]*$")
    set_property(GLOBAL PROPERTY JOB_POOLS sb_compile=${SB_ANDROID_BUILD_JOBS} sb_link=1)
    set(CMAKE_JOB_POOL_COMPILE sb_compile)
    set(CMAKE_JOB_POOL_LINK sb_link)
endif()
option(SB_ANDROID_COMPILE_CHECK "Compile runtime objects without game-derived code (never link/package)" OFF)
option(SB_ANDROID_DIAGNOSTICS "Build the game-free platform probe, not the game" OFF)
set(SB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(SPIRV-Tools_DIR "${SB_ROOT}/tools/android-arm64/lib/cmake/SPIRV-Tools" CACHE PATH "" FORCE)
set(SPIRV-Tools-opt_DIR "${SB_ROOT}/tools/android-arm64/lib/cmake/SPIRV-Tools-opt" CACHE PATH "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SB_ROOT}/tools/SDL" "${CMAKE_BINARY_DIR}/sdl")
if(SB_ANDROID_COMPILE_CHECK)
    include("${SB_ROOT}/scripts/android/compile_check.cmake")
endif()
if(SB_ANDROID_DIAGNOSTICS)
    # Standalone synthetic test, never linked into the APK or the game runtime.
    add_executable(android-bc-decode-test "${SB_ROOT}/tests/bc_decode_vulkan_test.cpp")
    target_include_directories(android-bc-decode-test PRIVATE "${SB_ROOT}/runtime")
    target_link_libraries(android-bc-decode-test PRIVATE vulkan)
    add_library(main SHARED "${SB_ROOT}/runtime/platform/android/diagnostics.cpp" "${SB_ROOT}/runtime/platform/android/physical_memory.cpp")
    target_link_libraries(main PRIVATE SDL3::SDL3 android log vulkan)
else()
    # The upstream root owns the recompilation/runtime targets. No alternate runtime.
    set(FFMPEG_XMA "${SB_ROOT}/tools/android-arm64/ffmpeg-xma" CACHE PATH "" FORCE)
    set(glslang_DIR "${SB_ROOT}/tools/android-arm64/lib/cmake/glslang" CACHE PATH "" FORCE)
    set(SDL3_DIR "${CMAKE_BINARY_DIR}/sdl" CACHE PATH "" FORCE)
    add_subdirectory("${SB_ROOT}" "${CMAKE_BINARY_DIR}/speedbreaker")
    # SDLActivity loads libmain.so and calls SDL_main.
    set_target_properties(SpeedBreaker PROPERTIES OUTPUT_NAME main)
    add_library(main ALIAS SpeedBreaker)
endif()
